
# Day 328 · printf 的整数与字符串支持

> 昨天解析了格式串,今天 Casey 实现具体的"打印整数"和"打印字符串"——把 FormatSpec 转成实际输出。包括数字到字符串的转换(itoa)、对齐填充、十六进制/八进制等。

## 0 · 为什么要有这一天

printf 的核心工作量在"打印整数"。`%d` 把 int 转成字符串,看着简单,但要处理:

1. **正负号**:负数加 `-`,正数看 flag 加 `+` 或空格
2. **进制**:`%d` 十进制,`%x` 十六进制,`%o` 八进制
3. **宽度**:`%10d` 至少 10 字符宽,不足填充空格或 0
4. **对齐**:`%-10d` 左对齐(默认右对齐)
5. **alt 形式**:`%#x` 前缀加 `0x`,`%#o` 前缀加 `0`

字符串 `%s` 相对简单——输出 + 对齐 + 精度(截断)。但 Casey 也要处理:

- **null 指针**:`%s` 传 NULL 应该输出 "(null)" 而不是崩
- **精度截断**:`%.5s` 只输出前 5 字符
- **宽字符**:`%ls` 是宽字符串(今天先不做)

今天 Casey 实现"打印函数",把 `FormatSpec + 参数` → `输出字符流`。这是手写 printf 的核心步骤。

心理锚点:今天之后你能"从零打印任意整数",理解 std::to_string 内部怎么工作。这是字符串处理的基础能力。

## 1 · Casey 今天做了什么(HH 脉络)

Casey 今天打开 `handmade_printf.cpp`,实现两个核心函数:

```c
void print_int(format_spec *spec, long long value, output_stream *out);
void print_string(format_spec *spec, const char *str, output_stream *out);
```

`print_int` 的核心算法:

1. **转字符串**:把数字转成字符串(支持十进制、十六进制、八进制)
2. **加符号**:负数 `-`,正数看 flag `+`/空格
3. **加前缀**:十六进制 `0x`(alt flag),八进制 `0`
4. **填充宽度**:宽度不够时填充空格(左/右)或零

```c
// 整数转字符串(十进制)
int itoa_dec(char *buf, long long val) {
    if (val == 0) { buf[0] = '0'; return 1; }
    
    bool negative = val < 0;
    unsigned long long uval = negative ? (unsigned long long)(-val) : (unsigned long long)val;
    
    char tmp[20];
    int len = 0;
    while (uval > 0) {
        tmp[len++] = '0' + (uval % 10);
        uval /= 10;
    }
    
    // 反转
    for (int i = 0; i < len; i++) {
        buf[i] = tmp[len - 1 - i];
    }
    return len;
}

// 十六进制
int itoa_hex(char *buf, unsigned long long val) {
    static const char digits[] = "0123456789abcdef";
    if (val == 0) { buf[0] = '0'; return 1; }
    
    char tmp[16];
    int len = 0;
    while (val > 0) {
        tmp[len++] = digits[val & 0xF];
        val >>= 4;
    }
    for (int i = 0; i < len; i++) buf[i] = tmp[len - 1 - i];
    return len;
}
```

`print_string` 更简单:

```c
void print_string(format_spec *spec, const char *str, output_stream *out) {
    if (str == NULL) str = "(null)";
    int len = strlen(str);
    if (spec->precision >= 0 && len > spec->precision) {
        len = spec->precision;
    }
    // 填充
    if (spec->width > len && !spec->left_align) {
        for (int i = 0; i < spec->width - len; i++) {
            out_putc(out, ' ');
        }
    }
    for (int i = 0; i < len; i++) {
        out_putc(out, str[i]);
    }
    if (spec->width > len && spec->left_align) {
        for (int i = 0; i < spec->width - len; i++) {
            out_putc(out, ' ');
        }
    }
}
```

Casey 在视频里强调几个工程细节:

1. **负数转无符号**:`-val` 在 `val = LLONG_MIN` 时溢出,要 `(unsigned long long)0 - (unsigned long long)val`
2. **buffer 大小**:64 位整数最多 20 字符(十进制)+ 符号 + 前缀,buffer 至少 24 字节
3. **避免递归**:数字转字符串用循环,不用递归(避免栈开销)

## 2 · 心智模型

### 类比:计数器显示

把整数打印想象成"计数器显示"。

计数器有 7 段显示管,显示 "0042"。你输入 42,显示要补 0 到 4 位。printf 的 `%04d` 就是"补 0 到指定宽度"。

进制转换类比"换算":42 = 4×10 + 2 = 0x2A = 052。printf 的 `%x`、`%o` 就是"换算到不同进制"。

### 严谨原理:整数转字符串

整数转字符串的核心算法(十进制):

```
val = 1234
buf = ""

while val > 0:
    digit = val % 10
    buf = char('0' + digit) + buf  // 前面插入
    val = val / 10

# buf = "1234"
```

但因为"前面插入"是 O(n),实际用"先反向填 + 最后反转":

```
val = 1234
tmp = []
i = 0

while val > 0:
    tmp[i] = '0' + (val % 10)  // '4', '3', '2', '1' 顺序
    i++
    val /= 10

# tmp = ['4', '3', '2', '1']
# 反转:tmp = ['1', '2', '3', '4']
```

为什么不用 sprintf?**避免递归**——sprintf 内部又要解析格式串,递归调用自己。Casey 的 printf 要自包含,不能依赖 sprintf。

### 十六进制转换

十六进制按 4 bit 分组:

```
val = 0xABCD = 43981

tmp = []
while val > 0:
    digit = val & 0xF       // 取低 4 bit
    tmp[i] = "0123456789abcdef"[digit]  // 查表
    i++
    val >>= 4               // 右移 4 bit

# tmp = ['D', 'C', 'B', 'A']
# 反转:tmp = ['A', 'B', 'C', 'D']
# 结果:"abcd"
```

`& 0xF` 和 `>>= 4` 比 `% 16` 和 `/= 16` 快(位运算 vs 除法),Casey 优先用位运算。

### 八进制类似

```
val = 0o777 = 511

while val > 0:
    digit = val & 0x7    // 取低 3 bit
    tmp[i] = '0' + digit
    val >>= 3
```

3 bit 一组,八进制。

### 填充与对齐

FormatSpec 的 width 字段决定"最小宽度",不足时要填充:

```
spec.width = 10
spec.zero_pad = true    // 填 0
spec.left_align = false  // 右对齐

数字 42 → "0000000042"(8 个 0 + "42")
```

实现:

```c
// 假设已经把数字转成 str(比如 "42"),长度 len=2
int padding = spec.width - len;
if (padding > 0) {
    if (!spec.left_align) {
        // 右对齐,先填充
        char pad = spec.zero_pad ? '0' : ' ';
        for (int i = 0; i < padding; i++) out_putc(out, pad);
        out_puts(out, str, len);  // 然后输出 str
    } else {
        // 左对齐,先 str 后填充
        out_puts(out, str, len);
        for (int i = 0; i < padding; i++) out_putc(out, ' ');
    }
} else {
    out_puts(out, str, len);
}
```

注意:**左对齐时不能填 0**(否则数字会变成 "4200000000")。这是 C99 标准规定——`-` 优先于 `0`。

### 符号和前缀

```c
// 符号
if (value < 0) {
    out_putc(out, '-');
} else if (spec.plus) {
    out_putc(out, '+');
} else if (spec.space) {
    out_putc(out, ' ');
}

// 前缀
if (spec.alt) {
    if (spec.specifier == 'x') {
        out_putc(out, '0');
        out_putc(out, 'x');
    } else if (spec.specifier == 'o') {
        out_putc(out, '0');
    }
}
```

`%+d` 显示正号,`% d` 正数前填空格,`%#x` 十六进制加 `0x` 前缀。这些 flag 修饰输出样式。

### Rust 实现

```rust
fn format_int(value: i64, base: u8, uppercase: bool, prefix: &str) -> String {
    if value == 0 {
        return format!("{}0", prefix);
    }
    
    let negative = value < 0;
    let uval: u64 = if negative {
        // 注意:i64::MIN 转 u64 不能简单 -val(溢出)
        value as u64
    } else {
        value as u64
    };
    
    let digits = if uppercase { "0123456789ABCDEF" } else { "0123456789abcdef" };
    let mut tmp = Vec::new();
    let mut v = uval;
    while v > 0 {
        let d = (v % base as u64) as usize;
        tmp.push(digits.as_bytes()[d] as char);
        v /= base as u64;
    }
    tmp.reverse();
    
    let mut result = String::new();
    if negative { result.push('-'); }
    result.push_str(prefix);
    result.extend(tmp);
    result
}
```

Rust 的 `i64 as u64` 是位转换,不溢出 —— `i64::MIN as u64 = 9223372036854775808`(直接转译位模式)。这避免了 C 的 `-LLONG_MIN` 溢出陷阱。

### 3.2 · 🐧 Linux 系统编程视角

Linux 内核的整数打印:

- **`%pK`**:内核指针(屏蔽 kptr_restrict)
- **`%pOF`**:device tree 节点
- **`%pI4`**:IPv4 地址(点分十进制)

`printk("%pI4", &ip)` 输出 `192.168.1.1`。这种"类型化指针"是内核特有,标准 printf 没有。

### 3.3 · 🦀 Rust 生态视角

Rust 的 Display trait 等价于 `%s`,Debug trait 等价于 `%?`。整数自动实现:

```rust
let n = 42;
println!("{}", n);        // 42
println!("{:?}", n);      // 42
println!("{:08}", n);     // 00000042
println!("{:#x}", n);     // 0x2a
println!("{:#o}", n);     // 0o52
println!("{:#b}", n);     // 0b101010
```

Rust 用 trait + format string,比 C printf 类型安全:

- `println!("{}", 42)` —— Display trait
- `println!("{:x}", 42)` —— LowerHex trait
- `println!("{:b}", 42)` —— Binary trait

每个类型可以自定义格式,扩展性强。

## 7 · 变式训练

### Lv1 · 概念辨析

**题**:`%-10d` 和 `%010d` 都填到 10 字符,差别在哪?

**参考解答**:

- `%-10d`:**左对齐**,填**空格**(左对齐时不能填 0)
- `%010d`:**右对齐**,填**0`(0` flag 表示填 0)

例:`%10d` 输出 42 → "        42"(8 空格 + 42,右对齐)
`%-10d` 输出 42 → "42        "(42 + 8 空格,左对齐)
`%010d` 输出 42 → "0000000042"(8 零 + 42,右对齐填 0)
`%-010d` 输出 42 → "42        "(`-` 优先,忽略 `0`)

### Lv2 · 动手实践

**题**:在 Rust 里实现完整的 `%d`、`%s`、`%x` 打印,包括 flags 和 width。

**参考解答**:

```rust
#[derive(Default, Debug)]
struct FormatSpec {
    left_align: bool,
    plus: bool,
    zero_pad: bool,
    alt: bool,
    width: usize,
    specifier: char,
}

fn format_int(spec: &FormatSpec, value: i64) -> String {
    let negative = value < 0;
    let uval: u64 = if negative { (value as u64).wrapping_neg() } else { value as u64 };
    
    // 数字转字符串
    let digits: String = if spec.specifier == 'x' {
        format!("{:x}", uval)
    } else if spec.specifier == 'o' {
        format!("{:o}", uval)
    } else {
        uval.to_string()
    };
    
    // 符号
    let sign = if negative { "-" } 
               else if spec.plus { "+" } 
               else { "" };
    
    // 前缀
    let prefix = if spec.alt {
        if spec.specifier == 'x' && uval != 0 { "0x" }
        else if spec.specifier == 'o' && !digits.starts_with('0') { "0" }
        else { "" }
    } else { "" };
    
    let content_len = sign.len() + prefix.len() + digits.len();
    let padding = if spec.width > content_len { spec.width - content_len } else { 0 };
    
    let mut result = String::new();
    if !spec.left_align && !spec.zero_pad {
        result.push_str(&" ".repeat(padding));
    }
    result.push_str(sign);
    result.push_str(prefix);
    if !spec.left_align && spec.zero_pad {
        result.push_str(&"0".repeat(padding));
    }
    result.push_str(&digits);
    if spec.left_align {
        result.push_str(&" ".repeat(padding));
    }
    result
}

fn format_string(spec: &FormatSpec, s: &str) -> String {
    let padding = if spec.width > s.len() { spec.width - s.len() } else { 0 };
    let mut result = String::new();
    if !spec.left_align {
        result.push_str(&" ".repeat(padding));
    }
    result.push_str(s);
    if spec.left_align {
        result.push_str(&" ".repeat(padding));
    }
    result
}

fn main() {
    let cases: &[(&str, i64)] = &[
        ("d", 42),
        ("d", -42),
        ("10d", 42),
        ("-10d", 42),
        ("010d", 42),
        ("+d", 42),
        ("x", 255),
        ("#x", 255),
        ("o", 8),
    ];
    
    for (fmt, val) in cases {
        let spec = FormatSpec {
            left_align: fmt.contains('-'),
            plus: fmt.contains('+'),
            zero_pad: fmt.contains('0') && !fmt.contains('-'),
            alt: fmt.contains('#'),
            width: fmt.matches(char::is_numeric).collect::<String>().parse().unwrap_or(0),
            specifier: fmt.chars().last().unwrap(),
        };
        let formatted = format_int(&spec, *val);
        println!("%{}  {} → [{}]", fmt, val, formatted);
    }
}
```

## 8 · Rust / Arch 落地代码

### Rust:完整 printf 整数 + 字符串(可跑)

```rust
// Cargo.toml: 无依赖
// src/main.rs

#[derive(Default, Debug, Clone)]
struct FormatSpec {
    left_align: bool,
    plus: bool,
    zero_pad: bool,
    alt: bool,
    space: bool,
    width: usize,
    precision: i32,
    specifier: char,
}

/// 整数转字符串(任意进制,2-16)
fn itoa_base(mut value: u64, base: u8, uppercase: bool) -> String {
    if value == 0 { return "0".to_string(); }
    let digits = if uppercase { "0123456789ABCDEF" } else { "0123456789abcdef" };
    let mut tmp = Vec::new();
    while value > 0 {
        let d = (value % base as u64) as usize;
        tmp.push(digits.as_bytes()[d]);
        value /= base as u64;
    }
    tmp.reverse();
    String::from_utf8(tmp).unwrap()
}

fn format_int(spec: &FormatSpec, value: i64) -> String {
    let negative = value < 0;
    // 处理 i64::MIN 不溢出
    let uval: u64 = if negative { 
        value.wrapping_neg() as u64 
    } else { 
        value as u64 
    };
    
    let digits = match spec.specifier {
        'x' => itoa_base(uval, 16, false),
        'X' => itoa_base(uval, 16, true),
        'o' => itoa_base(uval, 8, false),
        'b' => itoa_base(uval, 2, false),
        _ => itoa_base(uval, 10, false),
    };
    
    // 符号
    let sign: &str = if negative { "-" } 
                     else if spec.plus { "+" } 
                     else if spec.space { " " }
                     else { "" };
    
    // 前缀
    let prefix: String = if spec.alt {
        match spec.specifier {
            'x' if uval != 0 => "0x".to_string(),
            'X' if uval != 0 => "0X".to_string(),
            'o' if !digits.starts_with('0') => "0".to_string(),
            'b' if uval != 0 => "0b".to_string(),
            _ => String::new(),
        }
    } else { String::new() };
    
    let content_len = sign.len() + prefix.len() + digits.len();
    let padding = spec.width.saturating_sub(content_len);
    
    let mut result = String::new();
    // 右对齐 + 空格填充
    if !spec.left_align && !spec.zero_pad {
        result.push_str(&" ".repeat(padding));
    }
    result.push_str(sign);
    result.push_str(&prefix);
    // 右对齐 + 0 填充(在符号和前缀之后)
    if !spec.left_align && spec.zero_pad {
        result.push_str(&"0".repeat(padding));
    }
    result.push_str(&digits);
    // 左对齐
    if spec.left_align {
        result.push_str(&" ".repeat(padding));
    }
    result
}

fn format_string(spec: &FormatSpec, s: &str) -> String {
    let s = if s.is_empty() && spec.precision < 0 { s }
            else if spec.precision >= 0 {
                &s[..s.len().min(spec.precision as usize)]
            } else { s };
    let padding = spec.width.saturating_sub(s.len());
    let mut result = String::new();
    if !spec.left_align { result.push_str(&" ".repeat(padding)); }
    result.push_str(s);
    if spec.left_align { result.push_str(&" ".repeat(padding)); }
    result
}

fn main() {
    println!("=== Day 328: printf 整数 + 字符串 ===\n");
    
    // 整数测试
    println!("--- 整数 ---");
    let int_cases: &[(FormatSpec, i64, &str)] = &[
        (FormatSpec { specifier: 'd', ..Default::default() }, 42, "42"),
        (FormatSpec { specifier: 'd', ..Default::default() }, -42, "-42"),
        (FormatSpec { specifier: 'd', width: 10, ..Default::default() }, 42, "        42"),
        (FormatSpec { specifier: 'd', width: 10, left_align: true, ..Default::default() }, 42, "42        "),
        (FormatSpec { specifier: 'd', width: 10, zero_pad: true, ..Default::default() }, 42, "0000000042"),
        (FormatSpec { specifier: 'd', plus: true, ..Default::default() }, 42, "+42"),
        (FormatSpec { specifier: 'd', space: true, ..Default::default() }, 42, " 42"),
        (FormatSpec { specifier: 'x', ..Default::default() }, 255, "ff"),
        (FormatSpec { specifier: 'x', alt: true, ..Default::default() }, 255, "0xff"),
        (FormatSpec { specifier: 'X', ..Default::default() }, 255, "FF"),
        (FormatSpec { specifier: 'o', ..Default::default() }, 8, "10"),
        (FormatSpec { specifier: 'o', alt: true, ..Default::default() }, 8, "010"),
        (FormatSpec { specifier: 'b', ..Default::default() }, 10, "1010"),
    ];
    
    for (spec, val, expected) in int_cases {
        let actual = format_int(spec, *val);
        let status = if actual == *expected { "✓" } else { "✗" };
        println!("{} [{:>13}] = {:?} (expected {:?})", 
                 status, val, actual, expected);
    }
    
    // 字符串测试
    println!("\n--- 字符串 ---");
    let str_cases: &[(FormatSpec, &str, &str)] = &[
        (FormatSpec { specifier: 's', ..Default::default() }, "hello", "hello"),
        (FormatSpec { specifier: 's', width: 10, ..Default::default() }, "hi", "        hi"),
        (FormatSpec { specifier: 's', width: 10, left_align: true, ..Default::default() }, "hi", "hi        "),
        (FormatSpec { specifier: 's', precision: 3, ..Default::default() }, "hello", "hel"),
    ];
    
    for (spec, val, expected) in str_cases {
        let actual = format_string(spec, val);
        let status = if actual == *expected { "✓" } else { "✗" };
        println!("{} [{:>10}] = {:?} (expected {:?})", 
                 status, val, actual, expected);
    }
    
    // 边界情况
    println!("\n--- 边界 ---");
    println!("i64::MAX = {}", format_int(&FormatSpec { specifier: 'd', ..Default::default() }, i64::MAX));
    println!("i64::MIN = {}", format_int(&FormatSpec { specifier: 'd', ..Default::default() }, i64::MIN));
    println!("0       = {}", format_int(&FormatSpec { specifier: 'd', ..Default::default() }, 0));
}
```

## 10 · 深度补充:工程视角

整数转字符串的极致性能:Rust 的 itoa crate 用查表 + SIMD,5ns/int。C++ 的 fmtlib 类似。秘密:32 位整数分两半,每半 16 bit,查 2 字符的表(100 个条目),一次拼出。Casey 的简陋版用循环 50ns,工业版用 SIMD 5ns,差 10 倍。游戏 debug 输出 100k integers 时差别可观。
